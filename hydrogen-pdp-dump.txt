export const meta: Route.MetaFunction = ({data}) => {
  const product = data?.product;
  if (!product) return [{title: 'Product not found'}];
  const title = product.seo?.title || product.title;
  const description =
    product.seo?.description || product.description?.slice(0, 160) || '';
  const url = data.canonicalUrl;
  const image = product.featuredImage?.url;
  const variant = product.selectedOrFirstAvailableVariant;
  return [
    {title},
    {name: 'description', content: description},
    {tagName: 'link', rel: 'canonical', href: url},
    {property: 'og:type', content: 'product'},
    {property: 'og:title', content: title},
    {property: 'og:description', content: description},
    {property: 'og:url', content: url},
    ...(image ? [{property: 'og:image', content: image}] : []),
    {
      'script:ld+json': {
        '@context': 'https://schema.org',
        '@type': 'Product',
        name: product.title,
        description,
        url,
        ...(image ? {image: [image]} : {}),
        ...(variant?.sku ? {sku: variant.sku} : {}),
        brand: {'@type': 'Brand', name: product.vendor},
        ...(variant
          ? {
              offers: {
                '@type': 'Offer',
                url,
                priceCurrency: variant.price.currencyCode,
                price: variant.price.amount,
                availability: variant.availableForSale
                  ? 'https://schema.org/InStock'
                  : 'https://schema.org/OutOfStock',
              },
            }
          : {}),
      },
    },
  ];
};

export async function loader(args: Route.LoaderArgs) {
  // Start fetching non-critical data without blocking time to first byte
  const deferredData = loadDeferredData(args);

  // Await the critical data required to render initial state of the page
  const criticalData = await loadCriticalData(args);

  const productId = criticalData.product.id.split('/').pop() ?? '';
  const reviews = args.context.yotpo.getProductReviews(productId);
  return {...deferredData, ...criticalData, reviews};
}

/**
 * Load data necessary for rendering content above the fold. This is the critical data
 * needed to render the page. If it's unavailable, the whole page should 400 or 500 error.
 */
async function loadCriticalData({context, params, request}: Route.LoaderArgs) {
  const {handle} = params;
  const {storefront} = context;

  if (!handle) {
    throw new Error('Expected product handle to be defined');
  }

  const [{product}] = await Promise.all([
    storefront.query(PRODUCT_QUERY, {
      variables: {handle, selectedOptions: getSelectedProductOptions(request)},
    }),
    // Add other queries here, so that they are loaded in parallel
  ]);

  if (!product?.id) {
    throw new Response(null, {status: 404});
  }

  // The API handle might be localized, so redirect to the localized handle
  redirectIfHandleIsLocalized(request, {handle, data: product});

  return {
    canonicalUrl: new URL(request.url).origin + new URL(request.url).pathname,
    product,
    trustBadges: parseTrustBadges(product.trustBadges),
    specs: parseSpecs(product.specs),
    faqBody: product.faq?.reference?.body ?? null,
    techSpecsBody: product.techSpecs?.reference?.body ?? null,
    manualUrl: product.ownersManual?.reference?.url ?? null,
    assemblyVideoUrl: product.assemblyVideo?.value ?? null,
  };
}

/**
 * This is important to avoid re-fetching root queries on sub-navigations
 */
export const shouldRevalidate: ShouldRevalidateFunction = ({
  formMethod,
  currentUrl,
  nextUrl,
}) => {
  // revalidate when a mutation is performed e.g add to cart, login...
  if (formMethod && formMethod !== 'GET') return true;

  // revalidate when manually revalidating via useRevalidator
  if (currentUrl.toString() === nextUrl.toString()) return true;

  // Defaulting to no revalidation for root loader data to improve performance.
  // When using this feature, you risk your UI getting out of sync with your server.
  // Use with caution. If you are uncomfortable with this optimization, update the
  // line below to `return defaultShouldRevalidate` instead.
  // For more details see: https://remix.run/docs/en/main/route/should-revalidate
  return false;
};

/**

 * fetched after the initial page load. If it's unavailable, the page should still 200.
 * Make sure to not throw any errors here, as it will cause the page to 500.
 */

  const {storefront, customerAccount, cart} = context;

  // defer the footer query (below the fold)
  const footer = storefront
    .query(FOOTER_QUERY, {
      cache: storefront.CacheLong(),
      variables: {
        footerMenuHandle: 'footer', // Adjust to your footer menu handle
      },
    })
    .catch((error: Error) => {
      // Log query errors, but don't throw them so the page can still render
      console.error(error);
      return null;
    });
  return {
    cart: cart.get(),
    isLoggedIn: customerAccount.isLoggedIn(),
    footer,
  };
}

export function ErrorBoundary() {
  const error = useRouteError();
  let errorMessage: string | undefined;
  let errorStatus = 500;

  if (isRouteErrorResponse(error)) {
    errorMessage = typeof error.data === 'string' ? error.data : error.data?.message;
    errorStatus = error.status;
  } else if (error instanceof Error) {
    errorMessage = error.message;
  }

  return <ErrorBoundaryContent status={errorStatus} message={errorMessage} />;
}

  title: string;
  iconUrl?: string;
  iconAlt?: string;
  tooltip?: string;
  href?: string;
};

export type ProductSpec = {
  label: string;
  value: string;
  tooltip?: string;
  iconUrl?: string;
  iconAlt?: string;
  href?: string;
};

function fieldMap(fields: MetaField[] | undefined) {
  return Object.fromEntries((fields ?? []).map((f) => [f.key, f]));
}

function linkUrl(field: MetaField | undefined): string | undefined {
  if (!field?.value) return undefined;
  try {
    return (JSON.parse(field.value) as {url?: string}).url || undefined;
  } catch {
    return undefined;
  }
}

export function parseTrustBadges(metafield: MetaobjectList): TrustBadge[] {
  const badges: TrustBadge[] = [];
  for (const node of metafield?.references?.nodes ?? []) {
    const f = fieldMap(node.fields);
    const title = f.title?.value;
    if (!title) continue;
    const ref = f.icon?.reference;
    badges.push({
      title,
      iconUrl: ref?.image?.url ?? ref?.url ?? undefined,
      iconAlt: ref?.image?.altText ?? title,
      tooltip: f.tooltip?.value || undefined,
      href: linkUrl(f.link),
    });
  }
  return badges;
}

export function parseSpecs(metafield: MetaobjectList): ProductSpec[] {
  const specs: ProductSpec[] = [];
  for (const node of metafield?.references?.nodes ?? []) {
    const f = fieldMap(node.fields);
    const label = f.label?.value;
    const value = f.value?.value;
    if (!label || !value) continue;
    const ref = f.icon?.reference;
    specs.push({
      label,
      value,
      tooltip: f.tooltip?.value || undefined,
      iconUrl: ref?.image?.url ?? ref?.url ?? undefined,
      iconAlt: ref?.image?.altText ?? label,
      href: linkUrl(f.link),
    });
  }
  return specs;
}

import {useLocation} from 'react-router';
import type {SelectedOption} from '@shopify/hydrogen/storefront-api-types';
import {useMemo} from 'react';

export function useVariantUrl(
  handle: string,
  selectedOptions?: SelectedOption[],
) {
  const {pathname} = useLocation();

  return useMemo(() => {
    return getVariantUrl({
      handle,
      pathname,
      searchParams: new URLSearchParams(),
      selectedOptions,
    });
  }, [handle, selectedOptions, pathname]);
}

export function getVariantUrl({
  handle,
  pathname,
  searchParams,
  selectedOptions,
}: {
  handle: string;
  pathname: string;
  searchParams: URLSearchParams;
  selectedOptions?: SelectedOption[];
}) {
  const match = /(\/[a-zA-Z]{2}-[a-zA-Z]{2}\/)/g.exec(pathname);
  const isLocalePathname = match && match.length > 0;

  const path = isLocalePathname
    ? `${match![0]}products/${handle}`
    : `/products/${handle}`;

  selectedOptions?.forEach((option) => {
    searchParams.set(option.name, option.value);
  });

  const searchString = searchParams.toString();

  return path + (searchString ? '?' + searchParams.toString() : '');
}

/**
 * Creates Hydrogen context for React Router 7.9.x
 * Returns HydrogenRouterContextProvider with hybrid access patterns
 * */
export async function createHydrogenRouterContext(
  request: Request,
  env: Env,
  executionContext: ExecutionContext,
) {
  /**
   * Open a cache instance in the worker and a custom session instance.
   */
  if (!env?.SESSION_SECRET) {
    throw new Error('SESSION_SECRET environment variable is not set');
  }

  const waitUntil = executionContext.waitUntil.bind(executionContext);
  const [cache, session] = await Promise.all([
    caches.open('hydrogen'),
    AppSession.init(request, [env.SESSION_SECRET]),
  ]);

  const yotpo = createYotpoClient({env, cache, waitUntil, request});

  const hydrogenContext = createHydrogenContext(

      env,
      request,
      cache,
      waitUntil,
      session,
      // Or detect from URL path based on locale subpath, cookies, or any other strategy
      i18n: getLocaleFromRequest(request),
      cart: {
        queryFragment: CART_QUERY_FRAGMENT,
      },
    },

import type {I18nBase} from '@shopify/hydrogen';

export interface I18nLocale extends I18nBase {
  pathPrefix: string;
}

// Live locales. A prefix not listed here resolves to the default, so the
// guard in routes/($locale).tsx sees a mismatch and returns 404.
export const ENABLED_LOCALES = ['en-us'] as const;
export const DEFAULT_LOCALE = 'en-us';

export function getLocaleFromRequest(request: Request): I18nLocale {
  const first =
    new URL(request.url).pathname.split('/')[1]?.toLowerCase() ?? '';
  const locale = (ENABLED_LOCALES as readonly string[]).includes(first)
    ? first
    : DEFAULT_LOCALE;
  const [language, country] = locale.toUpperCase().split('-') as [
    I18nLocale['language'],
    I18nLocale['country'],
  ];
  return {language, country, pathPrefix: `/${locale}`};
}

const LOCALE_SHAPE = /^\/([a-z]{2}-[a-z]{2})(?=\/|$)/i;
const SKIP =
  /^\/(assets|graphiql|subrequest-profiler|__|@|node_modules|\.well-known)(\/|$)|^\/account\/authorize\/?$|\.[a-z0-9]+$/i;

// 308 unprefixed URLs to the default locale and mixed-case prefixes to lowercase.
// Unknown prefixes (e.g. /zz-zz) fall through so the 404 guard handles them.
export function localeRedirect(request: Request): Response | null {
  if (request.method !== 'GET' && request.method !== 'HEAD') return null;
  const url = new URL(request.url);
  const {pathname} = url;
  if (pathname.endsWith('/__manifest')) return null;
  if (SKIP.test(pathname)) return null;
  const match = pathname.match(LOCALE_SHAPE);
  let target: string | null = null;
  if (match) {
    const lower = match[1].toLowerCase();
    if (match[1] !== lower && (ENABLED_LOCALES as readonly string[]).includes(lower)) {
      target = '/' + lower + pathname.slice(match[0].length);
    }
  } else {
    target = `/${DEFAULT_LOCALE}${pathname === '/' ? '' : pathname}`;
  }
  if (!target) return null;
  return new Response(null, {
    status: 308,
    headers: {Location: target + url.search},
  });
}

# Design System

How the storefront's visual styles work: color, type, spacing, radius, shadow, layout, and the header's runtime contract. Read this before changing styles or the header.

Values come from `app/styles/tokens.css` and `app/styles/tailwind.css`. This replaces `docs/storefronts/design-system/design-system.md` in the old Next.js repo. `tokens.css` is identical to the old repo's `app/tokens.css`, and `tailwind.css` mirrors the old `globals.css` plus the fontsource imports.

> **Keeping this doc current:** if you change `tokens.css`, `tailwind.css`, or `components.json`, update this file in the same change.

## Where it lives

| File | Role |
| --- | --- |
| `app/styles/tokens.css` | Reference tokens: brand scales, spacing, type, borders, shadows, grid, logo, page blocks. Mobile values are the base; desktop overrides at `min-width: 768px`. Defines `.ds-container` and `.ds-grid`, `--font-heading`, `--font-body`. |
| `app/styles/tailwind.css` | Tailwind v4 entry. Imports tailwindcss, tw-animate-css, the fontsource packages and `tokens.css`. Holds the shadcn semantic variables and several `@theme inline` blocks that map variables to utilities. Defines `--font-geist-sans`, `--font-geist-mono`, `--font-fraunces`, `--text-xxs`. |
| `app/styles/reset.css`, `app.css` | Hydrogen template styles. Candidates for removal as the template UI is replaced. |
| `components.json` | shadcn config (style `base-nova`, neutral, CSS variables, lucide, `~` aliases, `rsc: false`). |
| `app/components/ui/` | Base components (base-ui primitives plus cva). |

## Header rules

- Use `LocalizedLink` for internal links. An unprefixed path costs a 308 redirect from `localeRedirect`.
- Never prefetch the account link. Sign-in is a plain `<a>` to the prefixed `/account/login` route (full navigation).
- Always render menu titles through `parseNavMenu`, on desktop and mobile. Never print raw `item.title`.
- Dropdowns need Escape to close, `aria-expanded` and `aria-haspopup` on triggers, and keyboard access to the panel.
- Do not start hide-on-scroll while keyboard focus is inside the header. Use `:has(:focus-visible)`, not `:focus-within`, which would pin the header open after a mouse click.
- Honor `prefers-reduced-motion` for looping or sliding animation.
- Text on `bg-card` must meet 4.5:1. `text-foreground/60` passes. `/50` and `/40` do not.
- No hardcoded English in new code once `use-intl` is added.

## Next.js to Hydrogen porting rules

- Replace `@/` imports with `~/`. Replace `next/link` and `next/image` with `LocalizedLink` and a plain `<img>` (or `remote-image`). Replace `next-intl` with the project's i18n once added.
- Remove `"use client"` and `"use server"` lines. Import from `react-router`, never `react-router-dom`.
- Keep Tailwind classes and tokens unchanged. Do not introduce raw hex, rgba, arbitrary pixel values or raw palette colors while porting.
- `components/product-detail/technical-specs.css` already uses only `var()` tokens and can be copied as is.

## Hydrogen specifics

- React 18.3, Hydrogen 2026.4.7. Do not copy React 19 code without checking.
- The CSP is generated in `entry.server.tsx`. Inline scripts need the nonce from `useNonce()`. External images and third-party scripts (Klaviyo, Algolia) must be added to the policy.
- Prefetch with `prefetch="intent"` only on safe, loader-only routes.

## Adding a token

1. Add the variable to `:root` in `app/styles/tokens.css` (desktop value in the `@media (min-width: 768px)` block if it scales).
2. Map it in an `@theme inline` block in `app/styles/tailwind.css`.
3. Use the generated utility, or `bg-(--name)` if unmapped.
4. Update the tables above.

## Adding a shadcn component

Run `pnpm dlx shadcn@latest add <name>`, then replace raw palette and pixel values in the generated file with semantic or `ds` tokens.

import {useState} from 'react';
import {Link, useNavigate} from 'react-router';
import {CartForm, type MappedProductOptions} from '@shopify/hydrogen';
import {useAside} from './Aside';
import type {ProductFragment} from 'storefrontapi.generated';

const pill = (selected: boolean, available: boolean, exists: boolean) =>
  [
    'grid place-items-center rounded-md border px-4 py-3 text-center text-base transition-colors',
    !exists || !available
      ? 'border-dashed border-neutral-300 text-neutral-400 line-through cursor-not-allowed'
      : selected
        ? 'border-black bg-black font-medium text-white'
        : 'border-neutral-300 bg-white text-neutral-700 hover:border-black hover:text-black',
  ].join(' ');

export function ProductForm({
  productOptions,
  selectedVariant,
}: {
  productOptions: MappedProductOptions[];
  selectedVariant: ProductFragment['selectedOrFirstAvailableVariant'];
}) {
  const navigate = useNavigate();
  const {open} = useAside();
  const [quantity, setQuantity] = useState(1);
  const soldOut = !selectedVariant || !selectedVariant.availableForSale;

  return (
    <div className="grid gap-6">
      {productOptions.map((option) => {
        if (option.optionValues.length === 1) return null;
        const selectedName = option.optionValues.find((v) => v.selected)?.name;
        return (
          <div key={option.name} className="grid gap-2.5">
            <p className="text-base font-semibold">
              {option.name}
              {selectedName ? (
                <span className="font-normal text-neutral-500">: {selectedName}</span>
              ) : null}
            </p>
            <div className="grid grid-cols-[repeat(auto-fit,minmax(6rem,1fr))] gap-2.5">
              {option.optionValues.map((value) => {
                const {name, handle, variantUriQuery, selected, available, exists, isDifferentProduct, swatch} = value;
                const color = swatch?.color;
                const image = swatch?.image?.previewImage?.url;
                const label = (
                  <span className="flex items-center justify-center gap-2">
                    {color || image ? (
                      <span
                        aria-hidden
                        className="size-5 shrink-0 rounded-full border border-black/20 bg-cover"
                        style={{backgroundColor: color || undefined, backgroundImage: image ? `url(${image})` : undefined}}
                      />
                    ) : null}
                    {name}
                  </span>
                );
                if (!exists || !available) {
                  return (
                    <span key={name} className={pill(selected, available, exists)}>
                      {label}
                    </span>
                  );
                }
                if (isDifferentProduct) {
                  return (
                    <Link
                      key={name}
                      prefetch="intent"
                      preventScrollReset
                      replace
                      to={`/products/${handle}?${variantUriQuery}`}
                      className={pill(selected, available, exists)}
                      aria-current={selected ? 'true' : undefined}
                    >
                      {label}
                    </Link>
                  );
                }
                return (
                  <button
                    key={name}
                    type="button"
                    aria-current={selected ? 'true' : undefined}
                    className={pill(selected, available, exists)}
                    onClick={() => {
                      if (!selected) {
                        void navigate(`?${variantUriQuery}`, {replace: true, preventScrollReset: true});
                      }
                    }}
                  >
                    {label}
                  </button>
                );
              })}
            </div>
          </div>
        );
      })}

      <div className="flex gap-2.5">
        <div
          role="group"
          aria-label="Item quantity"
          className="grid h-12 w-32 shrink-0 grid-cols-[3rem_2rem_3rem] rounded-md bg-white ring-1 ring-inset ring-neutral-300"
        >
          <button
            type="button"
            aria-label="Decrease quantity"
            disabled={quantity === 1}
            onClick={() => setQuantity((q) => Math.max(1, q - 1))}
            className="flex size-12 items-center justify-center disabled:cursor-not-allowed disabled:opacity-50"
          >
            −
          </button>
          <span role="status" aria-live="polite" className="flex h-12 w-8 items-center justify-center font-medium tabular-nums">
            {quantity}
          </span>
          <button
            type="button"
            aria-label="Increase quantity"
            disabled={quantity === 99}
            onClick={() => setQuantity((q) => Math.min(99, q + 1))}
            className="flex size-12 items-center justify-center disabled:cursor-not-allowed disabled:opacity-50"
          >
            +
          </button>
        </div>

        <div className="min-w-0 flex-1"><CartForm
          route="/cart"
          action={CartForm.ACTIONS.LinesAdd}
          inputs={{
            lines: selectedVariant
              ? [{merchandiseId: selectedVariant.id, quantity, selectedVariant}]
              : [],
          }}
        >
          {(fetcher) => (
            <button
              type="submit"
              disabled={soldOut || fetcher.state !== 'idle'}
              onClick={() => open('cart')}
              className="h-12 w-full min-w-0 flex-1 rounded-md bg-black font-semibold text-white hover:bg-black/85 disabled:cursor-not-allowed disabled:opacity-50"
            >
              {fetcher.state !== 'idle'
                ? 'Adding to Cart...'
                : soldOut
                  ? 'Out of Stock'
                  : 'Add to Cart'}
            </button>
          )}
        </CartForm></div>
      </div>
    </div>
  );
}

import {useEffect, useState} from 'react';
import {Image} from '@shopify/hydrogen';

type Img = {
  url: string;
  altText?: string | null;
  width?: number | null;
  height?: number | null;
};
type MediaNode = {
  mediaContentType: string;
  id?: string;
  image?: Img | null;
  previewImage?: Img | null;
  sources?: {url: string; mimeType: string}[] | null;
};

const thumbOf = (m: MediaNode): Img | null => m.image ?? m.previewImage ?? null;

// Remount with key={variantId} so the gallery jumps to the selected variant's image.
export function ProductGallery({
  media,
  title,
  activeUrl,
  className = '',
}: {
  media: MediaNode[];
  title: string;
  activeUrl?: string | null;
  className?: string;
}) {
  const items = media.filter(
    (m) => m.mediaContentType === 'IMAGE' || m.mediaContentType === 'VIDEO',
  );
  const initial = Math.max(
    0,
    items.findIndex((m) => thumbOf(m)?.url === activeUrl),
  );
  const [index, setIndex] = useState(initial);
  const [zoomed, setZoomed] = useState(false);
  const current = items[index] ?? items[0];

  useEffect(() => {
    if (!zoomed) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') setZoomed(false);
      if (e.key === 'ArrowRight') setIndex((i) => (i + 1) % items.length);
      if (e.key === 'ArrowLeft')
        setIndex((i) => (i - 1 + items.length) % items.length);
    };
    const prev = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    window.addEventListener('keydown', onKey);
    return () => {
      window.removeEventListener('keydown', onKey);
      document.body.style.overflow = prev;
    };
  }, [zoomed, items.length]);

  if (!current) {
    return <div className={`aspect-square rounded-lg bg-muted ${className}`} />;
  }
  const img = thumbOf(current);
  const isVideo = current.mediaContentType === 'VIDEO';

  return (
    <div className={`grid gap-3 ${className}`}>
      <div className="overflow-hidden rounded-lg bg-muted">
        {isVideo ? (
          <video
            key={current.id}
            controls
            playsInline
            poster={img?.url}
            className="aspect-square w-full object-cover"
          >
            {(current.sources ?? []).map((s) => (
              <source key={s.url} src={s.url} type={s.mimeType} />
            ))}
          </video>
        ) : img ? (
          <button
            type="button"
            onClick={() => setZoomed(true)}
            aria-label="Enlarge image"
            className="block h-full w-full cursor-zoom-in"
          >
            <Image
              key={current.id}
              data={{...img, altText: img.altText || title}}
              sizes="(min-width: 1024px) 60vw, 100vw"
              loading={index === 0 ? 'eager' : 'lazy'}
              className="aspect-square w-full object-cover"
            />
          </button>
        ) : null}
      </div>

      {items.length > 1 ? (
        <div className="flex gap-2 overflow-x-auto pb-1">
          {items.map((m, i) => {
            const t = thumbOf(m);
            if (!t) return null;
            return (
              <button
                key={m.id ?? i}
                type="button"
                onClick={() => setIndex(i)}
                aria-label={`Show ${m.mediaContentType === 'VIDEO' ? 'video' : 'image'} ${i + 1}`}
                aria-current={i === index}
                className={`relative size-20 shrink-0 overflow-hidden rounded-md ring-2 ${
                  i === index ? 'ring-foreground' : 'ring-transparent'
                }`}
              >
                <Image data={t} width={160} sizes="80px" className="size-full object-cover" />
                {m.mediaContentType === 'VIDEO' ? (
                  <span className="absolute inset-0 grid place-items-center bg-black/30 text-xs text-white">▶</span>
                ) : null}
              </button>
            );
          })}
        </div>
      ) : null}

      {zoomed ? (
        <div
          role="dialog"
          aria-modal="true"
          aria-label={`${title} enlarged`}
          className="fixed inset-0 z-[60] flex items-center justify-center bg-black/60 p-6 backdrop-blur-sm md:p-10"
          onClick={(e) => {
            if (e.target === e.currentTarget) setZoomed(false);
          }}
        >
          <button
            type="button"
            onClick={() => setZoomed(false)}
            aria-label="Close"
            className="absolute right-4 top-4 z-10 grid size-10 place-items-center rounded-full bg-black/50 text-xl text-white hover:opacity-80"
          >
            ×
          </button>
          {items.length > 1 ? (
            <>
              <button
                type="button"
                aria-label="Previous"
                onClick={() => setIndex((i) => (i - 1 + items.length) % items.length)}
                className="absolute left-4 top-1/2 z-10 grid size-10 -translate-y-1/2 place-items-center rounded-full bg-black/50 text-xl text-white hover:opacity-80"
              >
                ‹
              </button>
              <button
                type="button"
                aria-label="Next"
                onClick={() => setIndex((i) => (i + 1) % items.length)}
                className="absolute right-4 top-1/2 z-10 grid size-10 -translate-y-1/2 place-items-center rounded-full bg-black/50 text-xl text-white hover:opacity-80"
              >
                ›
              </button>
            </>
          ) : null}
          {isVideo ? (
            <video
              key={current.id}
              controls
              autoPlay
              playsInline
              poster={img?.url}
              className="max-h-full max-w-full object-contain"
            >
              {(current.sources ?? []).map((s) => (
                <source key={s.url} src={s.url} type={s.mimeType} />
              ))}
            </video>
          ) : img ? (
            <Image
              key={current.id}
              data={{...img, altText: img.altText || `${title} enlarged`}}
              sizes="90vw"
              loading="eager"
              className="max-h-full max-w-full bg-white object-contain"
            />
          ) : null}
        </div>
      ) : null}
    </div>
  );
}

import {useEffect, useRef, useState, type ReactNode} from 'react';

// Slides in at the top once the main buy block has scrolled off screen.
// Clicks the real Add to cart button so cart logic stays in one place.
export function StickyBuyBar({
  targetId,
  title,
  price,
  imageUrl,
}: {
  targetId: string;
  title: string;
  price: ReactNode;
  imageUrl?: string | null;
}) {
  const [visible, setVisible] = useState(false);
  const progress = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const update = () => {
      const el = document.getElementById(targetId);
      setVisible(!!el && el.getBoundingClientRect().bottom < 0);
      const max = document.documentElement.scrollHeight - window.innerHeight;
      const p = max > 0 ? Math.min(1, Math.max(0, window.scrollY / max)) : 0;
      if (progress.current) progress.current.style.transform = `scaleX(${p})`;
    };
    update();
    window.addEventListener('scroll', update, {passive: true});
    window.addEventListener('resize', update);
    return () => {
      window.removeEventListener('scroll', update);
      window.removeEventListener('resize', update);
    };
  }, [targetId]);

  const click = () =>
    document
      .querySelector<HTMLButtonElement>(`#${targetId} button:not([type="button"])`)
      ?.click();

  return (
    <div
      aria-hidden={!visible}
      {...({inert: visible ? undefined : ''} as object)}
      className={`fixed inset-x-0 top-0 z-40 bg-white transition-transform duration-200 ${
        visible ? 'translate-y-0' : '-translate-y-full'
      }`}
    >
      <div className="mx-auto flex max-w-7xl items-center justify-between gap-4 px-4 py-2">
        <div className="flex min-w-0 items-center gap-3">
          {imageUrl ? (
            <img src={imageUrl} alt="" width={56} height={56} className="size-14 shrink-0 object-contain" />
          ) : null}
          <span className="hidden min-w-0 truncate font-medium sm:block">{title}</span>
        </div>
        <div className="ml-auto flex items-center gap-4">
          <span className="text-xl font-bold tabular-nums">{price}</span>
          <button
            type="button"
            onClick={click}
            className="h-11 min-w-40 cursor-pointer rounded-md bg-black px-6 font-semibold text-white sm:min-w-56"
          >
            Add to cart
          </button>
        </div>
      </div>
      <div aria-hidden className="absolute inset-x-0 bottom-0 h-0.5 bg-black/10">
        <div ref={progress} className="h-full origin-left bg-black" style={{transform: 'scaleX(0)'}} />
      </div>
    </div>
  );
}

function formatRange(a: Date, b: Date) {
  const f = new Intl.DateTimeFormat('en-US', {month: 'short', day: 'numeric', timeZone: 'UTC'});
  if (a.getTime() === b.getTime()) return f.format(a);
  return f.formatRange(a, b);
}

              type="button"
              aria-label="Close"
              onClick={() => dialog.current?.close()}
              className="absolute right-0 top-0 hover:opacity-70"
            >
              ✕
            </button>
          </div>
          <p className="pt-5 text-lg font-bold">Update ZIP code</p>
          <p className="mb-4 mt-1">Enter your ZIP code to see your delivery date.</p>
          <div className="flex gap-3">
            <input
              inputMode="numeric"
              maxLength={5}
              value={draft}
              onChange={(e) => setDraft(e.target.value.replace(/\D/g, ''))}
              placeholder="ZIP code"
              aria-label="ZIP code"
              className="h-12 min-w-0 flex-1 rounded-md border px-3"
            />
            <button
              type="submit"
              disabled={draft.length !== 5}
              className="h-12 rounded-md bg-black px-6 font-medium text-white disabled:opacity-40"
            >
              Update
            </button>
          </div>
        </form>
      </dialog>
    </div>
  );
}

import {type FetcherWithComponents} from 'react-router';
import {CartForm, type OptimisticCartLineInput} from '@shopify/hydrogen';

export function AddToCartButton({
  analytics,
  children,
  disabled,
  lines,
  onClick,
}: {
  analytics?: unknown;
  children: React.ReactNode;
  disabled?: boolean;
  lines: Array<OptimisticCartLineInput>;
  onClick?: () => void;
}) {
  return (
    <CartForm route="/cart" inputs={{lines}} action={CartForm.ACTIONS.LinesAdd}>
      {(fetcher: FetcherWithComponents<any>) => (
        <>
          <input
            name="analytics"
            type="hidden"
            value={JSON.stringify(analytics)}
          />
          <button
            type="submit"
            onClick={onClick}
            disabled={disabled ?? fetcher.state !== 'idle'}
          >
            {children}
          </button>
        </>
      )}
    </CartForm>
  );
}
